**Route: GET /communities**

Apply `cacheMiddleware(TTL.COMMUNITIES)` as route-level middleware.

Query parameters:
- `?q=` (string, optional) — keyword search against name AND description via ILIKE
- `?sort=newest|popular` (string, optional, default: `newest`) — `newest` = `created_at DESC`, `popular` = `member_count DESC`
- `?cursor=` (string, optional) — base64url-encoded cursor from previous page
- `?limit=` (number, optional, default: 20, max: 100) — page size

Implementation:

```typescript
import { Router } from 'express';
import type { Request, Response } from 'express';
import { supabase } from '../lib/supabase.js';
import { cacheMiddleware, TTL } from '../middleware/cache.js';
import { encodeCursor, decodeCursor, parseLimit } from '../lib/utils.js';

export const communitiesRouter = Router();

communitiesRouter.get('/communities', cacheMiddleware(TTL.COMMUNITIES), async (req: Request, res: Response) => {
  const q = req.query.q as string | undefined;
  const sort = (req.query.sort as string) === 'popular' ? 'popular' : 'newest';
  const limit = parseLimit(req.query.limit as string | undefined, 20);
  const cursor = req.query.cursor as string | undefined;

  let query = supabase
    .schema('connect')
    .from('communities')
    .select('id, slug, name, description, member_count, thread_count, topic_id, slice_label, created_at');

  // Search: ILIKE on name OR description
  if (q) {
    query = query.or(`name.ilike.%${q}%,description.ilike.%${q}%`);
  }

  // Sort
  const sortColumn = sort === 'popular' ? 'member_count' : 'created_at';
  query = query.order(sortColumn, { ascending: false });
  query = query.order('id', { ascending: false }); // tiebreaker

  // Cursor (keyset pagination)
  if (cursor) {
    const decoded = decodeCursor(cursor);
    if (decoded) {
      query = query.or(
        `${sortColumn}.lt.${decoded.t},and(${sortColumn}.eq.${decoded.t},id.lt.${decoded.id})`
      );
    }
  }

  query = query.limit(limit + 1); // fetch one extra to determine hasMore

  const { data, error } = await query;

  if (error) {
    res.status(500).json({ error: { code: 'COMMUNITIES_FETCH_FAILED', message: error.message } });
    return;
  }

  const hasMore = (data?.length ?? 0) > limit;
  const items = (data ?? []).slice(0, limit).map(c => ({
    id: c.id,
    slug: c.slug,
    name: c.name,
    description: c.description,
    memberCount: c.member_count,
    threadCount: c.thread_count,
    topicId: c.topic_id,
    sliceLabel: c.slice_label,
    createdAt: c.created_at,
  }));

  const nextCursor = hasMore && items.length > 0
    ? encodeCursor(
        items[items.length - 1][sort === 'popular' ? 'memberCount' : 'createdAt'] as string,
        items[items.length - 1].id
      )
    : null;

  res.json({
    data: items,
    meta: { cursor: nextCursor, hasMore },
  });
});
```

IMPORTANT details:
- The `sort` parameter changes which column is used for BOTH ordering and cursor keyset — `created_at` for newest, `member_count` for popular
- For the `popular` sort, the cursor value `t` will be a number (member_count) rather than a timestamp string. The cursor encoding is opaque to the client so this is fine — decodeCursor returns `{ t: string, id: string }` and the Supabase `.lt()` filter coerces correctly for both types.
- The `.or()` filter for ILIKE search uses the supabase-js PostgREST syntax: `name.ilike.%term%,description.ilike.%term%`
- Response maps snake_case DB columns to camelCase API fields per CONTEXT.md

**Route: GET /communities/:id**

Apply `cacheMiddleware(TTL.COMMUNITIES)` as route-level middleware.

```typescript
communitiesRouter.get('/communities/:id', cacheMiddleware(TTL.COMMUNITIES), async (req: Request, res: Response) => {
  const { data, error } = await supabase
    .schema('connect')
    .from('communities')
    .select('id, slug, name, description, member_count, thread_count, topic_id, slice_label, created_at')
    .eq('id', req.params.id)
    .single();

  if (error || !data) {
    res.status(404).json({ error: { code: 'COMMUNITY_NOT_FOUND', message: 'Community not found' } });
    return;
  }

  res.json({
    data: {
      id: data.id,
      slug: data.slug,
      name: data.name,
      description: data.description,
      memberCount: data.member_count,
      threadCount: data.thread_count,
      topicId: data.topic_id,
      sliceLabel: data.slice_label,
      createdAt: data.created_at,
    },
  });
});
```

IMPORTANT: The `.single()` method returns an error (PGRST116) when zero or multiple rows match. Treat any error from `.single()` as a 404 — this handles both "not found" and "invalid UUID format" gracefully.

Do NOT add `updated_at` to the API response — it's internal metadata, not useful to API consumers.
