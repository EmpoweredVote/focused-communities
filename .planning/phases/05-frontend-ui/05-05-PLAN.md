    **useDraft (frontend/src/hooks/useDraft.ts):**
    Create a hook for localStorage draft preservation.

    - Parameters: `(key: string)` — e.g., `'draft-reply-${threadId}'` or `'draft-thread-${communityId}'`
    - Returns: `[value: string, setValue: (v: string) => void, clearDraft: () => void]`
    - On mount: read from `localStorage.getItem(key)`, use as initial state (or empty string if null)
    - On value change: `useEffect` that writes to `localStorage.setItem(key, value)` — debounced by 500ms to avoid excessive writes
    - `clearDraft`: removes the key from localStorage and resets state to empty string
    - If `key` changes (e.g., navigating to a different thread), re-read from localStorage

    Pattern:
    ```typescript
    export function useDraft(key: string): [string, (v: string) => void, () => void] {
      const [value, setValue] = useState(() => localStorage.getItem(key) ?? '')

      useEffect(() => {
        setValue(localStorage.getItem(key) ?? '')
      }, [key])

      useEffect(() => {
        const timer = setTimeout(() => {
          if (value) {
            localStorage.setItem(key, value)
          } else {
            localStorage.removeItem(key)
          }
        }, 500)
        return () => clearTimeout(timer)
      }, [key, value])

      const clearDraft = useCallback(() => {
        localStorage.removeItem(key)
        setValue('')
      }, [key])

      return [value, setValue, clearDraft]
    }
    ```
